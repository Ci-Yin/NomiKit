            modifier = Modifier
                .fillMaxWidth()
                .padding(bottom = 16.dp),
            verticalAlignment = Alignment.CenterVertically

            Text(
                text = "笔触宽度:",
                modifier = Modifier.padding(end = 8.dp)
            )

            Slider(
                value = currentStrokeWidth.value,
                onValueChange = { currentStrokeWidth = it.dp },
                valueRange = 1f..20f,
                modifier = Modifier.weight(1f)
            )

            Text(
                text = "${currentStrokeWidth.value.toInt()} dp",
                modifier = Modifier.padding(start = 8.dp)
            )
        }

        // 绘图区域
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .height(300.dp)
                .background(Color.White)
                .border(1.dp, Color.LightGray)
                // 绘制已保存的线条和当前线条
                .drawWithContent {
                    // 先绘制内容（如果有的话）
                    drawContent()

                    // 绘制已保存的线条
                    lines.forEach { line ->
                        drawLine(
                            color = line.color,
                            start = line.start,
                            end = line.end,
                            strokeWidth = line.strokeWidth.toPx()
                        )
                    }

                    // 绘制当前正在绘制的线条
                    currentLine?.let { line ->
                        drawLine(
                            color = line.color,
                            start = line.start,
                            end = line.end,
                            strokeWidth = line.strokeWidth.toPx()
                        )
                    }
                }
                // 处理触摸事件
                .pointerInput(Unit) {
                    detectDragGestures(
                        onDragStart = { offset ->
                            // 开始新的线条
                            currentLine = Line(
                                start = offset,
                                end = offset,
                                color = currentColor,
                                strokeWidth = currentStrokeWidth
                            )
                        },
                        onDrag = { change, dragAmount ->
                            change.consume()
                            // 更新当前线条的终点
                            currentLine?.let { line ->
                                currentLine = line.copy(end = line.end + dragAmount)
                            }
                        },
                        onDragEnd = {
                            // 保存当前线条
                            currentLine?.let { lines.add(it) }
                            currentLine = null
                        }
                    )
                }
        )

        // 清除按钮
        Button(
            onClick = { lines.clear() },
            modifier = Modifier
                .align(Alignment.End)
                .padding(top = 16.dp)
        ) {
            Text("清除")
        }
    }
}

@Composable
fun DrawWithAnimationExample() {
    // 动画状态
    var isPlaying by remember { mutableStateOf(false) }
    val progress by animateFloatAsState(
        targetValue = if (isPlaying) 1f else 0f,
        animationSpec = tween(2000, easing = FastOutSlowInEasing),
        label = "progress animation"
    )

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .padding(16.dp)
    ) {
        Text(
            text = "绘制与动画集成示例",
            style = MaterialTheme.typography.headlineSmall,
            modifier = Modifier.padding(bottom = 16.dp)
        )

        // 动画控制按钮
        Button(
            onClick = { isPlaying = !isPlaying },
            modifier = Modifier.align(Alignment.CenterHorizontally)
        ) {
            Text(if (isPlaying) "重置" else "播放")
        }

        Spacer(modifier = Modifier.height(16.dp))

        // 动画绘制区域
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .height(300.dp)
                .background(Color.White)
                .border(1.dp, Color.LightGray)
                .drawWithCache {
                    // 预计算
                    val width = size.width
                    val height = size.height
                    val center = Offset(width / 2, height / 2)
                    val radius = minOf(width, height) * 0.4f

                    // 创建渐变
                    val gradient = Brush.radialGradient(
                        colors = listOf(
                            Color(0xFF6200EE),
                            Color(0xFF3700B3)
                        ),
                        center = center,
                        radius = radius
                    )

                    onDrawBehind {
                        // 绘制背景
                        drawRect(Color(0xFFF5F5F5))

                        // 计算当前角度
                        val sweepAngle = 360f * progress

                        // 绘制进度圆弧
                        drawArc(
                            brush = gradient,
                            startAngle = -90f,
                            sweepAngle = sweepAngle,
                            useCenter = false,
                            topLeft = center - Offset(radius, radius),
                            size = Size(radius * 2, radius * 2),
                            style = Stroke(width = 20.dp.toPx(), cap = StrokeCap.Round)
                        )

                        // 绘制进度文本
                        val progressText = "${(progress * 100).toInt()}%"
                        val textPaint = Paint().asFrameworkPaint().apply {
                            textSize = 16.sp.toPx()
                            color = android.graphics.Color.BLACK
                            textAlign = android.graphics.Paint.Align.CENTER
                        }

                        drawContext.canvas.nativeCanvas.drawText(
                            progressText,
                            center.x,
                            center.y + textPaint.textSize / 3,
                            textPaint
                        )

                        // 绘制动画效果
                        if (progress > 0) {
                            // 计算当前点的位置
                            val angle = (-90f + sweepAngle) * (PI / 180f)
                            val x = center.x + cos(angle).toFloat() * radius
                            val y = center.y + sin(angle).toFloat() * radius
                            val currentPoint = Offset(x, y)

                            // 绘制当前点
                            drawCircle(
                                color = Color(0xFF6200EE),
                                radius = 10.dp.toPx(),
                                center = currentPoint
                            )

                            // 绘制光晕效果
                            for (i in 1..3) {
                                val alpha = 0.3f * (1 - i * 0.2f) * progress
                                drawCircle(
                                    color = Color(0xFF6200EE).copy(alpha = alpha),
                                    radius = (10 + i * 5).dp.toPx(),
                                    center = currentPoint
                                )
                            }
                        }
                    }
                }
        )
    }
}

@Composable
fun ParticleSystemExample() {
    // 粒子系统状态
    val particles = remember { mutableStateListOf<Particle>() }
    var isEmitting by remember { mutableStateOf(false) }
    var emitterPosition by remember { mutableStateOf(Offset.Zero) }

    // 粒子系统参数
    val particleCount = 100
    val particleLifetime = 2000 // 毫秒
    val gravity = Offset(0f, 0.1f) // 重力加速度

    // 动画定时器
    val infiniteTransition = rememberInfiniteTransition(label = "particle animation")
    val time by infiniteTransition.animateFloat(
        initialValue = 0f,
        targetValue = 1f,
        animationSpec = infiniteRepeatable(
            animation = tween(1000, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "time animation"
    )

    // 更新粒子状态
    LaunchedEffect(time, isEmitting) {
        // 移除过期粒子
        val currentTime = System.currentTimeMillis()
        particles.removeAll { particle ->
            currentTime - particle.birthTime > particleLifetime
        }

        // 更新现有粒子
        particles.forEachIndexed { index, particle ->
            // 应用重力和速度
            val newVelocity = particle.velocity + gravity
            val newPosition = particle.position + newVelocity

            // 更新粒子
            particles[index] = particle.copy(
                position = newPosition,
                velocity = newVelocity,
                alpha = 1f - (currentTime - particle.birthTime).toFloat() / particleLifetime
            )
        }

        // 如果正在发射，添加新粒子
        if (isEmitting && emitterPosition != Offset.Zero) {
            repeat(5) {
                if (particles.size < particleCount) {
                    // 创建随机方向的粒子
                    val angle = Random.nextDouble(0.0, 2 * PI).toFloat()
                    val speed = Random.nextDouble(1.0, 5.0).toFloat()
                    val velocity = Offset(
                        cos(angle) * speed,
                        sin(angle) * speed
                    )

                    // 随机颜色
                    val hue = Random.nextFloat() * 360f
                    val color = Color.hsl(hue, 0.8f, 0.7f)

                    // 添加新粒子
                    particles.add(
                        Particle(
                            position = emitterPosition,
                            velocity = velocity,
                            color = color,
                            size = Random.nextDouble(5.0, 15.0).toFloat().dp,
                            birthTime = currentTime,
                            alpha = 1f
                        )
                    )
                }
            }
        }
    }

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .padding(16.dp)
    ) {
        Text(
            text = "粒子系统效果",
            style = MaterialTheme.typography.headlineSmall,
            modifier = Modifier.padding(bottom = 16.dp)
        )

        // 粒子系统控制
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(bottom = 16.dp),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text("点击或拖动画布创建粒子")

            Switch(
                checked = isEmitting,
                onCheckedChange = { isEmitting = it }
            )
        }

        // 粒子系统画布
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .height(400.dp)
                .background(Color.Black)
                .drawWithContent {
                    // 绘制背景
                    drawContent()

                    // 绘制所有粒子
                    particles.forEach { particle ->
                        drawCircle(
                            color = particle.color.copy(alpha = particle.alpha),
                            radius = particle.size.toPx() / 2,
                            center = particle.position
                        )
                    }

                    // 绘制发射器
                    if (emitterPosition != Offset.Zero) {
                        drawCircle(
                            color = Color.White,
                            radius = 10.dp.toPx(),
                            center = emitterPosition
                        )
                    }
                }
                // 处理触摸事件
                .pointerInput(Unit) {
                    detectDragGestures(
                        onDragStart = { offset ->
                            emitterPosition = offset
                        },
                        onDrag = { change, _ ->
                            change.consume()
                            emitterPosition = change.position
                        },
                        onDragEnd = {
                            if (!isEmitting) {
                                emitterPosition = Offset.Zero
                            }
                        }
                    )
                }
                // 处理点击事件
                .pointerInput(Unit) {
                    detectTapGestures { offset ->
                        emitterPosition = offset
                        if (!isEmitting) {
                            // 点击时创建一组粒子
                            val currentTime = System.currentTimeMillis()
                            repeat(20) {
                                val angle = Random.nextDouble(0.0, 2 * PI).toFloat()
                                val speed = Random.nextDouble(2.0, 8.0).toFloat()
                                val velocity = Offset(
                                    cos(angle) * speed,
                                    sin(angle) * speed
                                )

                                val hue = Random.nextFloat() * 360f
                                val color = Color.hsl(hue, 0.8f, 0.7f)

                                particles.add(
                                    Particle(
                                        position = offset,
                                        velocity = velocity,
                                        color = color,
                                        size = Random.nextDouble(5.0, 15.0).toFloat().dp,
                                        birthTime = currentTime,
                                        alpha = 1f
                                    )
                                )
                            }
                        }
                    }
                }
        )
    }
}

@Composable
fun FluidAnimationExample() {
    // 流体动画状态
    val pointCount = 8
    val points = remember {
        List(pointCount) { index ->
            val angle = index * (2 * PI.toFloat() / pointCount)
            FluidPoint(
                basePosition = Offset(cos(angle), sin(angle)),
                animationPhase = index * (2 * PI.toFloat() / pointCount)
            )
        }
    }

    // 动画参数
    var amplitude by remember { mutableStateOf(0.2f) }
    var frequency by remember { mutableStateOf(1.5f) }
    var speed by remember { mutableStateOf(1.0f) }

    // 动画时间
    val infiniteTransition = rememberInfiniteTransition(label = "fluid animation")
    val time by infiniteTransition.animateFloat(
        initialValue = 0f,
        targetValue = 2 * PI.toFloat(),
        animationSpec = infiniteRepeatable(
            animation = tween(
                durationMillis = (2000 / speed).toInt(),
                easing = LinearEasing
            ),
            repeatMode = RepeatMode.Restart
        ),
        label = "time animation"
    )

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .padding(16.dp)
    ) {
        Text(
            text = "流体动画效果",
            style = MaterialTheme.typography.headlineSmall,
            modifier = Modifier.padding(bottom = 16.dp)
        )

        // 控制面板
        Card(
            modifier = Modifier
                .fillMaxWidth()
                .padding(bottom = 16.dp),
            elevation = CardDefaults.cardElevation(4.dp)
        ) {
            Column(
                modifier = Modifier.padding(16.dp)
            ) {
                // 振幅控制
                Text(
                    text = "振幅: ${(amplitude * 100).toInt()}%",
                    style = MaterialTheme.typography.bodyLarge
                )

                Slider(
                    value = amplitude,
                    onValueChange = { amplitude = it },
                    valueRange = 0f..0.5f,
                    modifier = Modifier.padding(vertical = 8.dp)
                )

                // 频率控制
                Text(
                    text = "频率: ${frequency.toInt()}x",
                    style = MaterialTheme.typography.bodyLarge
                )

                Slider(
                    value = frequency,
                    onValueChange = { frequency = it },
                    valueRange = 0.5f..5f,
                    modifier = Modifier.padding(vertical = 8.dp)
                )

                // 速度控制
                Text(
                    text = "速度: ${speed.toInt()}x",
                    style = MaterialTheme.typography.bodyLarge
                )

                Slider(
                    value = speed.toFloat(),
                    onValueChange = { speed = it.toDouble() },
                    valueRange = 0.5f..3f,
                    modifier = Modifier.padding(vertical = 8.dp)
                )
            }
        }

        // 流体动画展示
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .aspectRatio(1f)
                .background(Color(0xFF121212), RoundedCornerShape(16.dp))
                .padding(16.dp)
                .drawWithCache {
                    // 预计算
                    val width = size.width
                    val height = size.height
                    val center = Offset(width / 2, height / 2)
                    val radius = minOf(width, height) * 0.4f

                    // 创建渐变
                    val gradient = Brush.radialGradient(
                        colors = listOf(
                            Color(0xFF6200EE),
                            Color(0xFF3700B3)
                        ),
                        center = center,
                        radius = radius * 1.5f
                    )

                    onDrawBehind {
                        // 计算当前点的位置
                        val currentPoints = points.map { point ->
                            // 应用时间和频率
                            val currentPhase = point.animationPhase + time * frequency

                            // 计算当前半径变化
                            val radiusOffset = sin(currentPhase) * amplitude
                            val currentRadius = radius * (1 + radiusOffset)

                            // 计算当前位置
                            val x = center.x + point.basePosition.x * currentRadius
                            val y = center.y + point.basePosition.y * currentRadius

                            Offset(x, y)
                        }

                        // 创建流体路径
                        val path = Path()

                        // 移动到第一个点
                        path.moveTo(currentPoints[0].x, currentPoints[0].y)

                        // 使用贝塞尔曲线连接所有点
                        for (i in 0 until pointCount) {
                            val current = currentPoints[i]
                            val next = currentPoints[(i + 1) % pointCount]

                            // 计算控制点
                            val controlX = (current.x + next.x) / 2
                            val controlY = (current.y + next.y) / 2

                            // 添加曲线
                            path.quadraticBezierTo(
                                current.x, current.y,
                                controlX, controlY
                            )
                        }

                        // 闭合路径
                        path.close()

                        // 绘制流体形状
                        drawPath(
                            path = path,
                            brush = gradient
                        )

                        // 添加光晕效果
                        drawCircle(
                            color = Color.White.copy(alpha = 0.2f),
                            radius = radius * 0.8f,
                            center = center
                        )
                    }
                }
        )
    }
}

                    val height = size.height
                    val chartWidth = width
                    val chartHeight = height

                    // 计算最大值
                    val maxValue = data.maxOf { it.value }

                    // 计算每个条形的宽度和间距

                    val totalBarWidth = chartWidth * 0.8f

                    val barSpacing = chartWidth * 0.2f / (barCount + 1)

                        val y = chartHeight - (chartHeight * i / gridCount)
                        gridPath.moveTo(0f, y)
                        gridPath.lineTo(chartWidth, y)
                    }

                    onDrawBehind {
                        // 绘制网格线
                        drawPath(
                            path = gridPath,
                            color = Color.LightGray,
                            style = Stroke(
                                width = 1.dp.toPx(), pathEffect = PathEffect.dashPathEffect(
                                    floatArrayOf(5f, 5f), 0f
                                )
                            )
                        )

                        // 绘制Y轴标签
                        val textPaint = Paint().asFrameworkPaint().apply {
                            textSize = 10.sp.toPx()
                            color = android.graphics.Color.GRAY
                            textAlign = android.graphics.Paint.Align.RIGHT
                        }

                        for (i in 0..gridCount) {
                            val value = (maxValue * i / gridCount).toInt()
                            val y = chartHeight - (chartHeight * i / gridCount)

                            drawContext.canvas.nativeCanvas.drawText(
                                value.toString(),
                                -8.dp.toPx(),
                                y + textPaint.textSize / 3,
                                textPaint
                            )
                        }

                        // 绘制条形图
                        data.forEachIndexed { index, item ->
                            // 计算条形位置
                            val barHeight = (chartHeight * item.value / maxValue) * animatedProgress
                            val barLeft = barSpacing + index * (barWidth + barSpacing)
                            val barTop = chartHeight - barHeight

                            // 绘制条形
                            drawRect(
                                color = item.color,
                                topLeft = Offset(barLeft, barTop),
                                size = Size(barWidth, barHeight)
                            )

                            // 绘制数值标签
                            if (animatedProgress > 0.5f) {
                                val alpha = ((animatedProgress - 0.5f) * 2).coerceIn(0f, 1f)
                                val labelPaint = Paint().asFrameworkPaint().apply {
                                    textSize = 12.sp.toPx()
                                    color = android.graphics.Color.BLACK
                                    textAlign = android.graphics.Paint.Align.CENTER
                                    this.alpha = (255 * alpha).toInt()
                                }

                                drawContext.canvas.nativeCanvas.drawText(
                                    item.value.toInt().toString(),
                                    barLeft + barWidth / 2,
                                    barTop - 8.dp.toPx(),
                                    labelPaint
                                )
                            }

                            // 绘制X轴标签
                            val xLabelPaint = Paint().asFrameworkPaint().apply {
                                textSize = 10.sp.toPx()
                                color = android.graphics.Color.GRAY
                                textAlign = android.graphics.Paint.Align.CENTER
                            }

                            drawContext.canvas.nativeCanvas.drawText(
                                item.label,
                                barLeft + barWidth / 2,
                                chartHeight + 16.dp.toPx(),
                                xLabelPaint
                            )
                        }
                    }
                }
        )
    }
}

## 常见问题

### 1. 如何优化复杂绘制的性能？

**问题**：在实现复杂的绘制效果时，应用可能会出现卡顿或帧率下降，如何优化性能？

**解答**：优化复杂绘制的性能需要从多个方面入手：

1. **使用drawWithCache缓存计算结果**：
    - 将复杂的路径计算、对象创建等操作放在`drawWithCache`中
    - 只在必要时重新计算，避免每帧都重新创建对象

2. **减少对象分配**：
    - 重用对象，特别是集合和路径对象
    - 使用对象池模式管理频繁创建和销毁的对象
    - 避免在绘制回调中创建临时对象

3. **优化算法复杂度**：
    - 简化计算逻辑，使用更高效的算法
    - 对于大量数据，考虑采样或降低精度
    - 使用空间换时间的策略，预计算和缓存结果

4. **分层绘制**：
    - 将复杂UI分解为多个层，分别绘制
    - 只更新需要变化的层，保持其他层不变
    - 使用`graphicsLayer`应用硬件加速

5. **使用硬件加速**：
    - 尽可能使用GPU加速的绘制操作
    - 避免使用不支持硬件加速的API
    - 对于复杂效果，考虑使用Shader或RenderScript

6. **按需绘制**：
    - 只在可见区域绘制内容
    - 实现视口裁剪，忽略不可见部分
    - 对于滚动内容，实现虚拟化渲染

7. **优化状态管理**：
    - 避免不必要的状态更新触发重组
    - 使用`derivedStateOf`计算派生状态
    - 合理设计状态粒度，避免过度重绘

8. **使用性能分析工具**：
    - 使用Android Profiler分析性能瓶颈
    - 监测帧率和CPU/GPU使用情况
    - 使用Systrace分析渲染管线

通过综合应用这些优化策略，可以显著提高复杂绘制效果的性能，确保应用保持流畅的用户体验。

@Composable
fun CustomClockExercise() {
    // 状态
    var showDigital by remember { mutableStateOf(true) }
    var showSeconds by remember { mutableStateOf(true) }
    var use24Hour by remember { mutableStateOf(false) }

                        checked = showDigital,
                        onCheckedChange = { showDigital = it }

                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(bottom = 8.dp),
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.SpaceBetween
                ) {
                    Text("显示秒针")
                    Switch(
                        checked = showSeconds,
                        onCheckedChange = { showSeconds = it }

        // 时钟
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .aspectRatio(1f)
                .padding(16.dp)
                .drawWithCache {
                    // 预计算
                    val width = size.width
                    val height = size.height
                    val center = Offset(width / 2, height / 2)
                    val radius = minOf(width, height) / 2 * 0.9f

                    // 获取当前时间
                    val hour = currentTime.value.get(Calendar.HOUR)
                    val minute = currentTime.value.get(Calendar.MINUTE)
                    val second = currentTime.value.get(Calendar.SECOND)
                    val millis = currentTime.value.get(Calendar.MILLISECOND)

                    // 计算角度
                    val hourAngle = (hour * 30 + minute * 0.5f) * (PI / 180f)
                    val minuteAngle = minute * 6 * (PI / 180f)
                    val secondAngle = (second * 6 + millis * 0.006f) * (PI / 180f)

                    // 创建表盘路径
                    val dialPath = Path().apply {
                        addOval(Rect(center - Offset(radius, radius), center + Offset(radius, radius)))
                    }

                    // 创建刻度路径
                    val ticksPath = Path()
                    for (i in 0 until 60) {
                        val angle = i * 6 * (PI / 180f)
                        val sin = sin(angle).toFloat()
                        val cos = cos(angle).toFloat()

                        val isHour = i % 5 == 0
                        val tickStart = radius * (if (isHour) 0.8f else 0.9f)
                        val tickEnd = radius * 0.95f

                        val startX = center.x + cos * tickStart
                        val startY = center.y + sin * tickStart
                        val endX = center.x + cos * tickEnd
                        val endY = center.y + sin * tickEnd

                        ticksPath.moveTo(startX, startY)
                        ticksPath.lineTo(endX, endY)
                    }

                    // 创建数字路径和位置
                    val hourNumbers = mutableListOf<Pair<String, Offset>>()
                    for (i in 1..12) {
                        val angle = i * 30 * (PI / 180f)
                        val sin = sin(angle).toFloat()
                        val cos = cos(angle).toFloat()

                        val numberRadius = radius * 0.7f
                        val x = center.x + cos * numberRadius
                        val y = center.y + sin * numberRadius

                        hourNumbers.add(i.toString() to Offset(x, y))
                    }

                    onDrawBehind {
                        // 绘制表盘背景
                        drawCircle(
                            color = Color.White,
                            radius = radius,
                            center = center
                        )

                        // 绘制表盘边框
                        drawPath(
                            path = dialPath,
                            color = Color.Black,
                            style = Stroke(width = 2.dp.toPx())
                        )

                        // 绘制刻度
                        drawPath(
                            path = ticksPath,
                            color = Color.Black,
                            style = Stroke(width = 1.dp.toPx())
                        )

                        // 绘制数字

                            textSize = 16.sp.toPx()
                            color = android.graphics.Color.BLACK
                            textAlign = android.graphics.Paint.Align.CENTER
                        }

                        hourNumbers.forEach { (number, position) ->
                            drawContext.canvas.nativeCanvas.drawText(
                                number,
                                position.x,
                                position.y + textPaint.textSize / 3,
                                textPaint
                            )
                        }

                        // 绘制中心点
                        drawCircle(
                            color = Color.Black,
                            radius = 8.dp.toPx(),
                            center = center
                        )

                        // 绘制时针
                        val hourHandLength = radius * 0.5f
                        val hourHandEnd = Offset(
                            center.x + cos(hourAngle).toFloat() * hourHandLength,
                            center.y + sin(hourAngle).toFloat() * hourHandLength
                        )

                        drawLine(
                            color = Color.Black,
                            start = center,
                            end = hourHandEnd,
                            strokeWidth = 4.dp.toPx(),
                            cap = StrokeCap.Round
                        )

                        // 绘制分针
                        val minuteHandLength = radius * 0.7f
                        val minuteHandEnd = Offset(
                            center.x + cos(minuteAngle).toFloat() * minuteHandLength,
                            center.y + sin(minuteAngle).toFloat() * minuteHandLength
                        )

                        drawLine(
                            color = Color.Black,
                            start = center,
                            end = minuteHandEnd,
                            strokeWidth = 3.dp.toPx(),
                            cap = StrokeCap.Round
                        )

                        // 绘制秒针（如果启用）
                        if (showSeconds) {
                            val secondHandLength = radius * 0.8f
                            val secondHandEnd = Offset(
                                center.x + cos(secondAngle).toFloat() * secondHandLength,
                                center.y + sin(secondAngle).toFloat() * secondHandLength
                            )

                            drawLine(
                                color = Color.Red,
                                start = center,

@Composable
fun ParticleTextExercise() {
    // 状态
    var text by remember { mutableStateOf("COMPOSE") }
    var particleSize by remember { mutableStateOf(4.dp) }
    var particleSpacing by remember { mutableStateOf(6.dp) }
    var animationSpeed by remember { mutableStateOf(1f) }
    var isPlaying by remember { mutableStateOf(true) }

        listOf(
            Color(0xFF4285F4), // Google Blue
            Color(0xFFEA4335), // Google Red
            Color(0xFFFBBC05), // Google Yellow
            Color(0xFF34A853)  // Google Green
        )
    }

    // 动画状态
    val animationProgress = remember { Animatable(0f) }

        if (isPlaying) {
            animationProgress.animateTo(
                targetValue = 1f,
                animationSpec = infiniteRepeatable(
                    animation = tween(
                        durationMillis = (2000 / animationSpeed).toInt(),
                        easing = LinearEasing
                    ),
                    repeatMode = RepeatMode.Reverse
                )
            )
        } else {
            animationProgress.stop()
        }
    }

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .padding(16.dp)
    ) {
        Text(
            text = "练习3：粒子文字效果",
            style = MaterialTheme.typography.headlineSmall,

                Text(

                    style = MaterialTheme.typography.bodyLarge
                )

                Slider(
                    value = particleSize.value,
                    onValueChange = { particleSize = it.dp },
                    valueRange = 2f..10f,
                    modifier = Modifier.padding(vertical = 8.dp)
                )

                // 粒子间距
                Text(
                    text = "粒子间距: ${particleSpacing.value.toInt()} dp",
                    style = MaterialTheme.typography.bodyLarge
                )

                Slider(
                    value = particleSpacing.value,
                    onValueChange = { particleSpacing = it.dp },
                    valueRange = 4f..12f,
                    modifier = Modifier.padding(vertical = 8.dp)
                )

                // 动画速度
                Text(
                    text = "动画速度: ${animationSpeed.toInt()}x",
                    style = MaterialTheme.typography.bodyLarge
                )

                Slider(
                    value = animationSpeed,
                    onValueChange = { animationSpeed = it },
                    valueRange = 0.5f..3f,
                    modifier = Modifier.padding(vertical = 8.dp)
                )

                // 播放/暂停按钮
                Button(
                    onClick = { isPlaying = !isPlaying },
                    modifier = Modifier.align(Alignment.End)
                ) {
                    Text(if (isPlaying) "暂停" else "播放")
                }
            }
        }

        // 粒子文字效果
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .height(300.dp)
                .background(Color(0xFF121212), RoundedCornerShape(16.dp))
                .padding(16.dp),
            contentAlignment = Alignment.Center
        ) {
            // 粒子文字
            Canvas(
                modifier = Modifier.fillMaxSize()
            ) {
                // 计算文本尺寸
                val textPaint = Paint().asFrameworkPaint().apply {
                    textSize = 80.sp.toPx()
                    color = android.graphics.Color.WHITE
                    textAlign = android.graphics.Paint.Align.CENTER
                }

                val textWidth = textPaint.measureText(text)
                val textHeight = textPaint.fontMetrics.let { it.bottom - it.top }

                // 计算文本边界
                val textBounds = Rect()
                textPaint.getTextBounds(text, 0, text.length, textBounds)

                // 计算文本位置
                val textX = size.width / 2
                val textY = size.height / 2 + textBounds.height() / 2

                // 创建位图
                val bitmap = android.graphics.Bitmap.createBitmap(
                    size.width.toInt(),
                    size.height.toInt(),
                    android.graphics.Bitmap.Config.ARGB_8888
                )
                val canvas = android.graphics.Canvas(bitmap)

                // 在位图上绘制文本
                canvas.drawText(text, textX, textY, textPaint)

                // 粒子大小和间距
                val particleSizePx = particleSize.toPx()
                val particleSpacingPx = particleSpacing.toPx()

                // 扫描位图并创建粒子
                for (x in 0 until bitmap.width step particleSpacingPx.toInt()) {
                    for (y in 0 until bitmap.height step particleSpacingPx.toInt()) {
                        // 检查像素是否不透明
                        if (x < bitmap.width && y < bitmap.height) {
                            val pixel = bitmap.getPixel(x, y)
                            val alpha = android.graphics.Color.alpha(pixel)

                            if (alpha > 0) {
                                // 计算粒子目标位置
                                val targetX = x.toFloat()
                                val targetY = y.toFloat()

                                // 计算粒子当前位置（添加动画）
                                val progress = animationProgress.value
                                val amplitude = 20.dp.toPx() // 振幅

                                // 使用噪声函数创建随机运动

                                val noiseY = noise.noise(x * 0.05f, y * 0.05f, time + 100) * amplitude

                                val currentY = targetY + noiseY

                                // 选择颜色

                                val color = particleColors[colorIndex]

                                // 绘制粒子
                                drawCircle(
                                    color = color,
                                    radius = particleSizePx / 2,
                                    center = Offset(currentX, currentY)
                                )
                            }
                        }
                    }
                }

                // 释放资源
                bitmap.recycle()
            }
        }

// 简单的噪声函数实现
class SimplexNoise {
    fun noise(x: Float, y: Float, z: Float): Float {
        // 简化的噪声实现
        val dot = sin(x * 12.9898f + y * 78.233f + z * 37.719f) * 43758.5453f
        return (dot - floor(dot)).toFloat()
    }
}
