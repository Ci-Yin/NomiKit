## 概念介绍

在Jetpack
Compose中，ParentDataModifier是一种特殊的修饰符，它允许子组件向父组件传递信息，影响父组件对子组件的测量和放置行为。在上一节中，我们已经了解了ParentDataModifier的作用和常见类型，本节将深入探讨如何自己实现一个ParentDataModifier。

创建自定义ParentDataModifier的核心在于实现`ParentDataModifier`接口，并定义如何修改传递给父组件的数据。这个过程通常包括以下几个步骤：

1. **定义ParentData类**：创建一个数据类，包含你想传递给父组件的所有信息。
2. **实现ParentDataModifier接口**：创建一个实现`ParentDataModifier`接口的类，重写`modifyParentData`方法。
3. **为Modifier添加扩展函数**：创建一个扩展函数，使你的ParentDataModifier易于使用。
4. **在自定义布局中使用ParentData**：创建一个自定义布局，从子组件中获取ParentData并使用它来影响布局行为。

通过这种方式，你可以创建高度自定义的布局行为，使你的UI组件能够以特定的方式与父组件交互。这是Compose布局系统灵活性的一个重要体现，也是实现复杂UI效果的强大工具。

### 创建简单的ParentDataModifier

```kotlin
// 1. 定义ParentData类
data class PriorityParentData(
    val priority: Int = 0,
    val visible: Boolean = true
)

// 2. 实现ParentDataModifier接口
class PriorityModifier(
    private val priority: Int,
    private val visible: Boolean
) : ParentDataModifier {
    override fun Density.modifyParentData(parentData: Any?): Any {
        return (parentData as? PriorityParentData ?: PriorityParentData()).copy(
            priority = priority,
            visible = visible
        )
    }
}

// 3. 为Modifier添加扩展函数
fun Modifier.priority(
    priority: Int = 0,
    visible: Boolean = true
) = this.then(PriorityModifier(priority, visible))

// 4. 创建使用该ParentDataModifier的自定义布局
@Composable
fun PriorityLayout(
    modifier: Modifier = Modifier,
    content: @Composable () -> Unit
) {
    Layout(
        content = content,
        modifier = modifier
    ) { measurables, constraints ->
        // 过滤不可见的元素
        val visibleMeasurables = measurables.filter { measurable ->
            val parentData = measurable.parentData as? PriorityParentData
                ?: PriorityParentData()
            parentData.visible
        }

        // 按优先级排序
        val sortedMeasurables = visibleMeasurables.sortedByDescending { measurable ->
            val parentData = measurable.parentData as? PriorityParentData
                ?: PriorityParentData()
            parentData.priority
        }

        // 测量子元素
        val placeables = sortedMeasurables.map { measurable ->
            measurable.measure(constraints)
        }

        // 计算布局大小
        val width = placeables.maxOfOrNull { it.width } ?: 0
        val height = placeables.sumOf { it.height }

        // 放置子元素
        layout(width, height) {
            var y = 0
            placeables.forEach { placeable ->
                placeable.place(0, y)
                y += placeable.height
            }
        }
    }
}
```

```kotlin
// 1. 定义更复杂的ParentData类
data class LayoutParentData(
    val alignment: Alignment = Alignment.TopStart,
    val weight: Float = 0f,
    val spacing: Dp = 0.dp,
    val visible: Boolean = true
)

// 2. 实现ParentDataModifier接口
class LayoutModifier(
    private val alignment: Alignment,
    private val weight: Float,
    private val spacing: Dp,
    private val visible: Boolean
) : ParentDataModifier {
    override fun Density.modifyParentData(parentData: Any?): Any {
        return when (parentData) {
            is LayoutParentData -> parentData.copy(
                alignment = alignment,
                weight = weight,
                spacing = spacing,
                visible = visible
            )
            else -> LayoutParentData(
                alignment = alignment,
                weight = weight,
                spacing = spacing,
                visible = visible
            )
        }
    }
}

// 3. 为Modifier添加扩展函数
fun Modifier.layoutParams(
    alignment: Alignment = Alignment.TopStart,
    weight: Float = 0f,
    spacing: Dp = 0.dp,
    visible: Boolean = true
) = this.then(
    LayoutModifier(
        alignment = alignment,
        weight = weight,
        spacing = spacing,
        visible = visible
    )
)

// 4. 创建使用该ParentDataModifier的自定义布局
@Composable
fun FlexLayout(
    modifier: Modifier = Modifier,
    orientation: Orientation = Orientation.Vertical,
    content: @Composable () -> Unit
) {
    Layout(
        content = content,
        modifier = modifier
    ) { measurables, constraints ->
        // 过滤不可见的元素
        val visibleMeasurables = measurables.filter { measurable ->
            val parentData = measurable.parentData as? LayoutParentData
                ?: LayoutParentData()
            parentData.visible
        }

        // 计算权重和间距
        val totalWeight = visibleMeasurables.sumOf { measurable ->
            val parentData = measurable.parentData as? LayoutParentData
                ?: LayoutParentData()
            parentData.weight.toDouble()
        }.toFloat()

        // 计算固定大小的元素所需空间
        val fixedMeasurables = visibleMeasurables.filter { measurable ->
            val parentData = measurable.parentData as? LayoutParentData
                ?: LayoutParentData()
            parentData.weight == 0f
        }

        // 计算权重元素
        val weightMeasurables = visibleMeasurables.filter { measurable ->
            val parentData = measurable.parentData as? LayoutParentData
                ?: LayoutParentData()
            parentData.weight > 0f
        }

        // 计算间距总和
        val totalSpacing = visibleMeasurables.sumOf { measurable ->
            val parentData = measurable.parentData as? LayoutParentData
                ?: LayoutParentData()
            parentData.spacing.roundToPx().toDouble()
        }

        // 根据方向计算布局
        val (width, height, placeables) = if (orientation == Orientation.Vertical) {
            // 测量固定高度的元素
            val fixedPlaceables = fixedMeasurables.map { measurable ->
                measurable.measure(constraints.copy(minHeight = 0))
            }

            // 计算剩余空间
            val fixedHeight = fixedPlaceables.sumOf { it.height }
            val remainingHeight = constraints.maxHeight - fixedHeight - totalSpacing.toInt()
            val weightUnit = if (totalWeight > 0f) remainingHeight / totalWeight else 0f

            // 测量权重元素
            val weightPlaceables = weightMeasurables.map { measurable ->
                val parentData = measurable.parentData as? LayoutParentData
                    ?: LayoutParentData()
                val weight = parentData.weight
                val height = (weightUnit * weight).roundToInt()

                measurable.measure(
                    constraints.copy(
                        minHeight = height,
                        maxHeight = height
                    )
                )
            }

            // 计算总宽度和高度
            val allPlaceables = fixedPlaceables + weightPlaceables
            val width = allPlaceables.maxOfOrNull { it.width } ?: 0
            val height = allPlaceables.sumOf { it.height } + totalSpacing.toInt()

            Triple(width, height, allPlaceables)
        } else {
            // 测量固定宽度的元素
            val fixedPlaceables = fixedMeasurables.map { measurable ->
                measurable.measure(constraints.copy(minWidth = 0))
            }

            // 计算剩余空间
            val fixedWidth = fixedPlaceables.sumOf { it.width }
            val remainingWidth = constraints.maxWidth - fixedWidth - totalSpacing.toInt()
            val weightUnit = if (totalWeight > 0f) remainingWidth / totalWeight else 0f

            // 测量权重元素
            val weightPlaceables = weightMeasurables.map { measurable ->
                val parentData = measurable.parentData as? LayoutParentData
                    ?: LayoutParentData()
                val weight = parentData.weight
                val width = (weightUnit * weight).roundToInt()

                measurable.measure(
                    constraints.copy(
                        minWidth = width,
                        maxWidth = width
                    )
                )
            }

            // 计算总宽度和高度
            val allPlaceables = fixedPlaceables + weightPlaceables
            val width = allPlaceables.sumOf { it.width } + totalSpacing.toInt()
            val height = allPlaceables.maxOfOrNull { it.height } ?: 0

            Triple(width, height, allPlaceables)
        }

        // 创建布局
        layout(width, height) {
            // 放置子元素
            var x = 0
            var y = 0

            placeables.forEachIndexed { index, placeable ->
                val measurable = visibleMeasurables[index]
                val parentData = measurable.parentData as? LayoutParentData
                    ?: LayoutParentData()

                // 根据对齐方式计算位置
                val xPos = if (orientation == Orientation.Vertical) {
                    when (parentData.alignment) {
                        Alignment.TopStart, Alignment.CenterStart, Alignment.BottomStart -> 0
                        Alignment.TopEnd, Alignment.CenterEnd, Alignment.BottomEnd -> width - placeable.width
                        else -> (width - placeable.width) / 2 // Center
                    }
                } else {
                    x
                }

                val yPos = if (orientation == Orientation.Horizontal) {
                    when (parentData.alignment) {
                        Alignment.TopStart, Alignment.TopCenter, Alignment.TopEnd -> 0
                        Alignment.BottomStart, Alignment.BottomCenter, Alignment.BottomEnd -> height - placeable.height
                        else -> (height - placeable.height) / 2 // Center
                    }
                } else {
                    y
                }

                // 放置元素
                placeable.place(xPos, yPos)

                // 更新位置
                if (orientation == Orientation.Vertical) {
                    y += placeable.height + parentData.spacing.roundToPx()
                } else {
                    x += placeable.width + parentData.spacing.roundToPx()
                }
            }
        }
    }
}
```

```kotlin
    y: Int = 0,
    zIndex: Float = 0f

// 4. 创建处理多种ParentData的自定义布局
@Composable
fun CanvasLayout(
    modifier: Modifier = Modifier,
    content: @Composable () -> Unit
) {
    Layout(
        content = content,
        modifier = modifier
    ) { measurables, constraints ->
        // 测量所有子元素
        val placeables = measurables.map { measurable ->
            // 获取位置数据
            val positionData = measurable.parentData as? PositionParentData
                ?: PositionParentData()

            // 获取样式数据
            val styleData = measurable.parentData as? StyleParentData
                ?: StyleParentData()

            // 测量子元素
            val placeable = measurable.measure(constraints.copy(minWidth = 0, minHeight = 0))

            // 返回测量结果和相关数据
            Triple(placeable, positionData, styleData)
        }

        // 计算布局大小
        val width = constraints.maxWidth
        val height = constraints.maxHeight

        // 创建布局
        layout(width, height) {
            // 按Z轴索引排序并放置子元素
            placeables
                .sortedBy { (_, positionData, _) -> positionData.zIndex }
                .forEach { (placeable, positionData, styleData) ->
                    // 放置子元素，应用位置和样式
                    placeable.placeWithLayer(
                        x = positionData.x,
                        y = positionData.y
                    ) {
                        // 应用样式
                        scaleX = styleData.scale
                        scaleY = styleData.scale
                        alpha = styleData.alpha
                        rotationZ = styleData.rotation
                    }
                }
        }
    }
}
```

## 进阶用法

### 创建可组合的ParentDataModifier

```kotlin
// 1. 定义可组合的ParentData类
interface ComposableParentData

data class AlignmentData(val alignment: Alignment) : ComposableParentData
data class WeightData(val weight: Float) : ComposableParentData
data class SpacingData(val spacing: Dp) : ComposableParentData
data class VisibilityData(val visible: Boolean) : ComposableParentData

// 2. 创建组合ParentData容器
data class CompositeParentData(
    val dataMap: Map<Class<out ComposableParentData>, ComposableParentData> = emptyMap()
) {
    // 获取特定类型的数据
    inline fun <reified T : ComposableParentData> getData(): T? {
        return dataMap[T::class.java] as? T
    }

    // 添加或更新数据
    fun with(data: ComposableParentData): CompositeParentData {
        val newMap = dataMap.toMutableMap()
        newMap[data::class.java] = data
        return CompositeParentData(newMap)
    }
}

// 3. 实现可组合的ParentDataModifier
abstract class ComposableParentDataModifier : ParentDataModifier {
    abstract fun createData(): ComposableParentData

    override fun Density.modifyParentData(parentData: Any?): Any {
        val compositeData = when (parentData) {
            is CompositeParentData -> parentData
            else -> CompositeParentData()
        }

        return compositeData.with(createData())
    }
}

// 4. 实现具体的修饰符
class AlignmentModifier(private val alignment: Alignment) : ComposableParentDataModifier() {
    override fun createData(): ComposableParentData = AlignmentData(alignment)
}

class WeightModifier(private val weight: Float) : ComposableParentDataModifier() {
    override fun createData(): ComposableParentData = WeightData(weight)
}

class SpacingModifier(private val spacing: Dp) : ComposableParentDataModifier() {
    override fun createData(): ComposableParentData = SpacingData(spacing)
}

class VisibilityModifier(private val visible: Boolean) : ComposableParentDataModifier() {
    override fun createData(): ComposableParentData = VisibilityData(visible)
}

// 5. 为Modifier添加扩展函数
fun Modifier.itemAlignment(alignment: Alignment) = this.then(AlignmentModifier(alignment))
fun Modifier.itemWeight(weight: Float) = this.then(WeightModifier(weight))
fun Modifier.itemSpacing(spacing: Dp) = this.then(SpacingModifier(spacing))
fun Modifier.itemVisibility(visible: Boolean) = this.then(VisibilityModifier(visible))

// 6. 创建使用组合ParentData的自定义布局
@Composable
fun ComposableLayout(
    modifier: Modifier = Modifier,
    orientation: Orientation = Orientation.Vertical,
    content: @Composable () -> Unit
) {
    Layout(
        content = content,
        modifier = modifier
    ) { measurables, constraints ->
        // 过滤不可见的元素
        val visibleMeasurables = measurables.filter { measurable ->
            val compositeData = measurable.parentData as? CompositeParentData
                ?: CompositeParentData()
            val visibilityData = compositeData.getData<VisibilityData>()
            visibilityData?.visible ?: true
        }

        // 计算权重和间距
        val weightMeasurables = mutableListOf<Pair<Measurable, Float>>()
        val fixedMeasurables = mutableListOf<Measurable>()

        visibleMeasurables.forEach { measurable ->
            val compositeData = measurable.parentData as? CompositeParentData
                ?: CompositeParentData()
            val weightData = compositeData.getData<WeightData>()

            if (weightData != null && weightData.weight > 0f) {
                weightMeasurables.add(measurable to weightData.weight)
            } else {
                fixedMeasurables.add(measurable)
            }
        }

        // 计算总权重
        val totalWeight = weightMeasurables.sumOf { it.second.toDouble() }.toFloat()

        // 计算间距
        val spacingMap = visibleMeasurables.associateWith { measurable ->
            val compositeData = measurable.parentData as? CompositeParentData
                ?: CompositeParentData()
            val spacingData = compositeData.getData<SpacingData>()
            spacingData?.spacing ?: 0.dp
        }

        val totalSpacing = spacingMap.values.sumOf { it.roundToPx() }

        // 根据方向计算布局
        val (width, height, placeables) = if (orientation == Orientation.Vertical) {
            // 测量固定高度的元素
            val fixedPlaceables = fixedMeasurables.map { measurable ->
                measurable.measure(constraints.copy(minHeight = 0))
            }

            // 计算剩余空间
            val fixedHeight = fixedPlaceables.sumOf { it.height }
            val remainingHeight = (constraints.maxHeight - fixedHeight - totalSpacing).coerceAtLeast(0)
            val weightUnit = if (totalWeight > 0f) remainingHeight / totalWeight else 0f

            // 测量权重元素
            val weightPlaceables = weightMeasurables.map { (measurable, weight) ->
                val height = (weightUnit * weight).roundToInt()
                measurable.measure(
                    constraints.copy(
                        minHeight = height,
                        maxHeight = height
                    )
                )
            }

            // 计算总宽度和高度
            val allPlaceables = fixedPlaceables + weightPlaceables
            val width = allPlaceables.maxOfOrNull { it.width } ?: 0
            val height = allPlaceables.sumOf { it.height } + totalSpacing

            Triple(width, height, allPlaceables)
        } else {
            // 测量固定宽度的元素
            val fixedPlaceables = fixedMeasurables.map { measurable ->
                measurable.measure(constraints.copy(minWidth = 0))
            }

            // 计算剩余空间
            val fixedWidth = fixedPlaceables.sumOf { it.width }
            val remainingWidth = (constraints.maxWidth - fixedWidth - totalSpacing).coerceAtLeast(0)
            val weightUnit = if (totalWeight > 0f) remainingWidth / totalWeight else 0f

            // 测量权重元素
            val weightPlaceables = weightMeasurables.map { (measurable, weight) ->
                val width = (weightUnit * weight).roundToInt()
                measurable.measure(
                    constraints.copy(
                        minWidth = width,
                        maxWidth = width
                    )
                )
            }

            // 计算总宽度和高度
            val allPlaceables = fixedPlaceables + weightPlaceables
            val width = allPlaceables.sumOf { it.width } + totalSpacing
            val height = allPlaceables.maxOfOrNull { it.height } ?: 0

            Triple(width, height, allPlaceables)
        }

        // 创建布局
        layout(width, height) {
            // 放置子元素
            var x = 0
            var y = 0

            placeables.zip(visibleMeasurables).forEach { (placeable, measurable) ->
                val compositeData = measurable.parentData as? CompositeParentData
                    ?: CompositeParentData()

                // 获取对齐方式
                val alignmentData = compositeData.getData<AlignmentData>()
                val alignment = alignmentData?.alignment ?: Alignment.TopStart

                // 获取间距
                val spacingData = compositeData.getData<SpacingData>()
                val spacing = spacingData?.spacing?.roundToPx() ?: 0

                // 根据对齐方式计算位置
                val xPos = if (orientation == Orientation.Vertical) {
                    when (alignment) {
                        Alignment.TopStart, Alignment.CenterStart, Alignment.BottomStart -> 0
                        Alignment.TopEnd, Alignment.CenterEnd, Alignment.BottomEnd -> width - placeable.width
                        else -> (width - placeable.width) / 2 // Center
                    }
                } else {
                    x
                }

                val yPos = if (orientation == Orientation.Horizontal) {
                    when (alignment) {
                        Alignment.TopStart, Alignment.TopCenter, Alignment.TopEnd -> 0
                        Alignment.BottomStart, Alignment.BottomCenter, Alignment.BottomEnd -> height - placeable.height
                        else -> (height - placeable.height) / 2 // Center
                    }
                } else {
                    y
                }

                // 放置元素
                placeable.place(xPos, yPos)

                // 更新位置
                if (orientation == Orientation.Vertical) {
                    y += placeable.height + spacing
                } else {
                    x += placeable.width + spacing
                }
            }
        }
    }
}
```

```kotlin
                    scaleY = parentData.scale
                    alpha = parentData.alpha
                    rotationZ = parentData.rotation
```

### 1. 如何处理多个ParentDataModifier的冲突？

**问题**：当我在同一个组件上应用多个ParentDataModifier时，它们可能会相互覆盖。如何正确处理这种情况？

**解答**：处理多个ParentDataModifier的冲突需要理解修饰符的应用顺序和覆盖规则：

1. **了解修饰符的应用顺序**：
   修饰符从左到右（或从上到下）依次应用，后面的修饰符可能会覆盖前面的修饰符的效果。

   ```kotlin
   // 在这个例子中，align(Alignment.BottomEnd)会覆盖align(Alignment.TopStart)
   Box {
       Text(
           "文本",
           modifier = Modifier
               .align(Alignment.TopStart)
               .align(Alignment.BottomEnd)
       )
   }
   ```

2. **使用组合ParentData**：
   创建一个能够组合多种不同类型ParentData的容器类，如前面示例中的`CompositeParentData`。

   ```kotlin
   data class CompositeParentData(
       val dataMap: Map<Class<out ComposableParentData>, ComposableParentData> = emptyMap()
   ) {
       // 获取特定类型的数据
       inline fun <reified T : ComposableParentData> getData(): T? {
           return dataMap[T::class.java] as? T
       }

       // 添加或更新数据
       fun with(data: ComposableParentData): CompositeParentData {
           val newMap = dataMap.toMutableMap()
           newMap[data::class.java] = data
           return CompositeParentData(newMap)
       }
   }
   ```

3. **实现智能的modifyParentData方法**：
   在`modifyParentData`方法中，根据现有的ParentData类型进行不同的处理。

   ```kotlin
               parentData.copy(

                   // 保留其他属性
                   property2 = parentData.property2

           is OtherParentData -> {
               // 转换为MyParentData，尝试保留相关信息
               MyParentData(
                   property1 = newValue1,
                   property2 = convertFromOther(parentData)
               )
           }
           else -> {
               // 创建新的ParentData
               MyParentData(
                   property1 = newValue1,
                   property2 = defaultValue2
               )
           }
       }
   }
   ```

### 2. 如何优化ParentDataModifier的性能？

**问题**：在复杂的UI中，大量使用ParentDataModifier可能会影响性能。有哪些方法可以优化ParentDataModifier的性能？

**解答**：优化ParentDataModifier的性能需要从多个方面入手：

1. **减少对象创建**：
   每次调用`modifyParentData`方法时，避免不必要的对象创建。

   ```kotlin
   // 不好的做法：总是创建新对象
   override fun Density.modifyParentData(parentData: Any?): Any {
       return MyParentData(property = value)
   }

   // 更好的做法：复用现有对象
   override fun Density.modifyParentData(parentData: Any?): Any {
       return if (parentData is MyParentData && parentData.otherProps == defaultOtherProps) {
           parentData.copy(property = value)
       } else {
           MyParentData(property = value)
       }
   }
   ```

2. **使用不可变数据类**：
   使用Kotlin的数据类来表示ParentData，利用其内置的`copy`方法高效地创建修改后的实例。

   ```kotlin
   data class MyParentData(
       val prop1: Int = 0,
       val prop2: String = "",
       val prop3: Boolean = false
   )

   // 高效地修改部分属性
   val newData = oldData.copy(prop1 = newValue)
   ```

3. **缓存计算结果**：
   在布局过程中，缓存基于ParentData的计算结果，避免重复计算。

   ```kotlin

               val parentData = measurable.parentData as? MyParentData

                   // 计算布局结果

               }
           }

           // 使用缓存的结果进行布局
           // ...
       }
   }

   // 缓存类
   class LayoutCache {
       private val cache = mutableMapOf<CacheKey, LayoutResult>()

       fun getOrCompute(key: CacheKey, compute: () -> LayoutResult): LayoutResult {
           return cache.getOrPut(key, compute)
       }

       // 在适当的时候清理缓存

           cache.clear()
       }
   }
   ```

4. **延迟计算**：
   只在实际需要时才执行基于ParentData的计算。

   ```kotlin
   Layout(
       content = content,
       modifier = modifier
   ) { measurables, constraints ->
       // 先收集所有ParentData
       val parentDataList = measurables.map { it.parentData as? MyParentData ?: MyParentData() }

       // 检查是否需要复杂计算
       val needsComplexLayout = parentDataList.any { it.requiresComplexLayout }

       // 根据需要选择布局策略
       val layoutStrategy = if (needsComplexLayout) {
           ComplexLayoutStrategy()
       } else {
           SimpleLayoutStrategy()
       }

       // 使用选定的策略进行布局
       layoutStrategy.layout(measurables, parentDataList, constraints)
   }
   ```

5. **批量处理**：
   一次性处理多个子元素的ParentData，而不是单独处理每个元素。

   ```kotlin
   // 批量计算布局参数
   val layoutParams = measurables.map { measurable ->
       val parentData = measurable.parentData as? MyParentData ?: MyParentData()
       computeLayoutParams(parentData)
   }

   // 一次性分配空间
   val availableSpace = constraints.maxWidth
   val allocatedWidths = distributeSpace(layoutParams, availableSpace)

   // 使用分配的空间测量子元素
   val placeables = measurables.zip(allocatedWidths).map { (measurable, width) ->
       measurable.measure(constraints.copy(minWidth = width, maxWidth = width))
   }
   ```

6. **避免不必要的重组**：
   确保ParentDataModifier的变化不会触发不必要的重组。

   ```kotlin
   // 使用remember缓存修饰符
   val itemModifier = remember(priority, visible) {
       Modifier.priority(priority, visible)
   }

   // 使用派生状态避免不必要的更新
   val derivedModifier = remember {
       derivedStateOf {
           Modifier.priority(calculatePriority(state), isVisible(state))
       }
   }
   ```

7. **优化数据结构**：
   使用高效的数据结构来存储和检索ParentData。

   ```kotlin
   // 使用IntMap代替普通Map，当键是整数时
   class OptimizedParentData(
       private val properties: IntMap<Any> = IntMap()
   ```

8. **使用原始类型**：
   对于简单的ParentData，考虑使用原始类型而不是对象。

   ```kotlin
   // 不好的做法：使用对象

   // 更好的做法：直接使用Float
   ```

通过这些优化技术，你可以显著提高ParentDataModifier的性能，特别是在复杂UI和大量子元素的情况下。关键是减少对象创建、避免重复计算，并使用高效的数据结构。

   ```kotlin
   fun AnimatableLayout(
       modifier: Modifier = Modifier,
       content: @Composable () -> Unit

       Layout(
           content = content,
           modifier = modifier
       ) { measurables, constraints ->
           // 测量子元素
           val placeables = measurables.map { measurable ->
               val parentData = measurable.parentData as? AnimatableParentData
                   ?: AnimatableParentData()

               measurable.measure(constraints) to parentData

           // 计算布局大小
           val width = constraints.maxWidth
           val height = constraints.maxHeight

           // 创建布局
           layout(width, height) {
               // 放置子元素
               placeables.forEach { (placeable, parentData) ->
                   // 计算位置
                   val x = (width - placeable.width) / 2 + parentData.offset.x.roundToInt()
                   val y = (height - placeable.height) / 2 + parentData.offset.y.roundToInt()

                   // 应用变换
                   placeable.placeWithLayer(x, y) {
                       scaleX = parentData.scale
                       scaleY = parentData.scale
                       alpha = parentData.alpha
                       rotationZ = parentData.rotation
                       // 注意：颜色变换需要特殊处理
                   }
               }
           }
       }
   }
   ```

```kotlin
        layout(width, height) {
            // 放置子元素
            placeables.forEach { (placeable, parentData) ->
                // 根据对齐方式计算位置
                val x = when (parentData.alignment) {
                    Alignment.TopStart, Alignment.CenterStart, Alignment.BottomStart -> 0
                    Alignment.TopEnd, Alignment.CenterEnd, Alignment.BottomEnd -> width - placeable.width
                    else -> (width - placeable.width) / 2 // Center
                }

                val y = when (parentData.alignment) {
                    Alignment.TopStart, Alignment.TopCenter, Alignment.TopEnd -> 0
                    Alignment.BottomStart, Alignment.BottomCenter, Alignment.BottomEnd -> height - placeable.height
                    else -> (height - placeable.height) / 2 // Center
                }

                // 放置元素
                placeable.place(x, y)
            }
        }
    }
}
```

```kotlin
@Composable
fun CalendarLayout(
    modifier: Modifier = Modifier,
    onDateClick: (LocalDate) -> Unit = {},
    content: @Composable () -> Unit
) {
    Layout(
        content = content,
        modifier = modifier
    ) { measurables, constraints ->
        // 确保有足够的子元素（7列 x 6行 = 42个日期单元格）
        if (measurables.size < 42) {
            throw IllegalArgumentException("日历布局需要42个子元素（6行7列）")
        }

        // 计算单元格大小
        val cellWidth = constraints.maxWidth / 7
        val cellHeight = constraints.maxHeight / 6

        // 测量所有子元素
        val placeables = measurables.map { measurable ->
            val cellConstraints = Constraints.fixed(cellWidth, cellHeight)
            measurable.measure(cellConstraints)
        }

        // 创建布局
        layout(constraints.maxWidth, constraints.maxHeight) {
            // 放置子元素
            placeables.forEachIndexed { index, placeable ->
                val row = index / 7
                val col = index % 7

                val x = col * cellWidth
                val y = row * cellHeight

                placeable.place(x, y)
            }
        }
    }
}

// 辅助函数：生成日历天数
fun generateCalendarDays(yearMonth: YearMonth): List<LocalDate> {
    val firstDayOfMonth = yearMonth.atDay(1)
    val lastDayOfMonth = yearMonth.atEndOfMonth()

    // 计算第一天是星期几（0是星期日，1是星期一，以此类推）
    val dayOfWeek = firstDayOfMonth.dayOfWeek.value % 7

    // 计算上个月需要显示的天数
    val daysFromPreviousMonth = if (dayOfWeek == 0) 0 else dayOfWeek

    // 生成日历天数列表
    val calendarDays = mutableListOf<LocalDate>()

    // 添加上个月的日期
    if (daysFromPreviousMonth > 0) {
        val previousMonth = yearMonth.minusMonths(1)
        val lastDayOfPreviousMonth = previousMonth.atEndOfMonth()

        for (i in daysFromPreviousMonth downTo 1) {
            calendarDays.add(lastDayOfPreviousMonth.minusDays(i - 1L))
        }
    }

    // 添加当前月的日期
    for (i in 1..lastDayOfMonth.dayOfMonth) {
        calendarDays.add(yearMonth.atDay(i))
    }

    // 添加下个月的日期，直到填满42个格子
    val remainingDays = 42 - calendarDays.size
    if (remainingDays > 0) {
        val nextMonth = yearMonth.plusMonths(1)

        for (i in 1..remainingDays) {
            calendarDays.add(nextMonth.atDay(i))
        }
    }

    return calendarDays
}

// 辅助函数：获取中文星期几
fun getChineseDayOfWeek(dayOfWeek: DayOfWeek): String {
    return when (dayOfWeek) {
        DayOfWeek.MONDAY -> "一"
        DayOfWeek.TUESDAY -> "二"
        DayOfWeek.WEDNESDAY -> "三"
        DayOfWeek.THURSDAY -> "四"
        DayOfWeek.FRIDAY -> "五"
        DayOfWeek.SATURDAY -> "六"
        DayOfWeek.SUNDAY -> "日"
    }
}
```
